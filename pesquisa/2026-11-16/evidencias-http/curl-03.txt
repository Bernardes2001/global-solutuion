function exemplo() {
	var url = "https://" + "sistemas.tcepe.tc.br:443/DadosAbertos" 
	url = url + "/" + $("#idEntidade").val() + "!" + $("#idFormato").val();
	var parametros = "";
	var parametrosObrigatoriosNaoPreenchidos = "";
	$("input[name='valoresExemplo']").each(function(index) {
		if($(this).val() != null && $(this).val().trim() != "") {
			if(parametros != "") {
				parametros = parametros + "&";
			}
			parametros = parametros + $(this).attr('id') + "=" + $(this).val().trim();
		} else if($(this).val() != null && $(this).val().trim() == "" && $(this).next().val() == "S") {
			if(parametrosObrigatoriosNaoPreenchidos != "") {
				parametrosObrigatoriosNaoPreenchidos = parametrosObrigatoriosNaoPreenchidos + "\n";
			}
			parametrosObrigatoriosNaoPreenchidos = parametrosObrigatoriosNaoPreenchidos + $(this).attr('id');
		}
	});
	if(parametrosObrigatoriosNaoPreenchidos != "") {
		alert("Parâmetro(s) obrigatório(s) não preenchido(s): " + parametrosObrigatoriosNaoPreenchidos);
	} else if(parametros != "") {
		window.open(url + "?" + parametros);
	} else {
		window.open(url);
	}
}
</script>
